} catch (DateTimeParseException ex) {
    throw new DateTimeParseException(
            String.format(
                "Invalid value for field [%s]: raw='%s', errorIndex=%d, "
                    + "expected format yyyy-MM-dd'T'HH:mm:ss.SSS'Z'. Cause: %s",
                PROCESSING_DATE, rawValue, ex.getErrorIndex(), ex.getMessage()),
            rawValue,             // getParsedString()
            ex.getErrorIndex(),   // getErrorIndex()
            ex);                  // original exception kept as cause
}
