MSG="chore: add configuration and set base image versions before stable release"

for d in */; do
  [ -d "$d/.git" ] || continue
  echo "== $d"
  git -C "$d" add -A
  if git -C "$d" diff --cached --quiet; then
    echo "   nothing to commit"
    continue
  fi
  git -C "$d" commit -m "$MSG" && \
  git -C "$d" push -o ci.skip -u origin HEAD
done
