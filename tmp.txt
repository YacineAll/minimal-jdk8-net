return {
  "phelipetls/jsonpath.nvim",
  lazy = false,
  config = function()
    vim.api.nvim_create_autocmd("FileType", {
      pattern = { "json", "jsonc" },
      callback = function()
        vim.opt_local.winbar = "%{%v:lua.require'jsonpath'.get()%}"
      end,
    })

    vim.keymap.set("n", "<leader>jp", function()
      vim.fn.setreg("+", require("jsonpath").get())
    end, { desc = "Copy JSON path" })
  end,
}
